# Make all variable names lowercase
names(data) <- tolower(names(data))

data$poc <- rowSums(data[, c("black", "americanindian", "asianamerican", "latine", "mena")])

data$racsum <- rowSums(data[, c("white", "black", "americanindian", "asianamerican", "latine", "mena")])

# Subset the data, where if POC then keep, otherwise remove.
data <- data %>%
  filter(poc > 0)

Demographic Table

Show the code 
# First we create a subset of data that we want a summary of.
selected_vars <- c("sex", "gender", "age", "education", "racem", "income", "sexualidentity", "urbanicity")
dem_data <- subset(data, select = selected_vars)

# We need to convert labelled data into factors. 
dem_data <- dem_data %>%
  mutate(across(c(sex, gender, education, income, racem, sexualidentity, urbanicity), ~ haven::as_factor(.)))

dem_data$age <- as.numeric(dem_data$age)

dem_table <- gtsummary::tbl_summary(
    dem_data,
    include = c(age, sex, gender, racem, sexualidentity, education, income, urbanicity),
    missing = "no",
    type = (list(
      age ~ "continuous2")
    ),
        statistic = list(
      all_continuous() ~ "{mean} ({sd})",
      all_categorical() ~ "{n} ({p}%)"
    ),
    digits = list(
      all_continuous() ~ 2,
      all_categorical() ~ c(0, 2)),
    label = list(
      age ~ "Age",
    sex ~ "Sex Assigned at Birth",
    gender ~ "Gender Identity",
    racem ~ "Racial Identity",
    sexualidentity ~ "Sexual Orientation",
    income ~ "Financial Hardship",
    education ~ "Education",
    urbanicity ~ "Residence"))

dem_table %>%
  as_flex_table() %>%
